git status -- mostra o status do que precisa ser add ou commit
git add (nome ou pasta do arquivo) -- envia aquele arquivo para adicionar
git add . -- Envia todos os arquivos
git commit -- envia os arquivos editados git
git commit (nome do arquivo) -m "mensagem" -- envia o arquivo com mensagem 
git commit -a -- envia todos os arquivos
git push -- Envia todos os arquivos que foram commitados para o github
git pull -- Baixa todos os arquivos do repositorio que foram atualizados
git clone -- Clona o projeto do repositor
git rm (nome arquivo) --
git log -- Faz log do que tem sido feito no git -- Teclar a tecla "Q" para sair do log
git mv -- pode renomear os arquivos ou mudar para outra pasta exemplo git mv rodape.css(local do arquivo) css/rodape.css(aonde vai o arquivo)
git checkout -- pega o arquivo que inicia antes de qualquer alteração
git reset --hard origem/master -- reset de tudo que foi feito
gitignore -- fazer um arquivo chamado .gitignore e escrever o nome dos arquivos que serão ignorados.
git branch -- verificar os branchs disponiveis
git branch "Nome do branch" -- Cria um novo branch
git branch -d ou --delete -- deleta um branch
git checkout <nome> -- muda para o branch já criado
git checkout -b <nome> -- muda o e cria um branch que estamos (colocar o nome do branch)
git merger <nome> -- unifica 2 branchs
git stash -- recomeça mas salva onde voce pode recuperar depois
git stash list -- lista as stash que tem
git stash apply <id stash> -- recupera a stash
git stash show -p <id stash> -- mostra o que foi alterado
git stash clear -- limpa totalmente as stash
git stash drop <nome> -- deleta uma stash específica
git tag -a <nome> -m<msg> -- criar tags nos branchs, é um checkpoint de um branch
git show <nome tag> -- verifica o detalhe da tags para sair usar a letra Q
git push origin <nome da tag> -- tags podem ser enviadas para o repositorio
git push origin --tags -- envia todas as tags
git fetch -- mapea o branch de outros users
git remote add origin <link> -- criamos repo remoto
